# 1- Crie um programa que deve funcionar como um conversor de temperaturas. Ele deve conter um menu que permita ao usuário selecionar 2 opções:
# Celsius para Fahrenheit
# Fahrenheit para Celsius
# Com base na escolha do usuário, seu programa deve perguntar a ele a temperatura que deseja converter, calcular, e mostrar o resultado na tela para o usuário.
# Fórmulas: 
# Fahrenheit = (Celsius * 9/5) + 32
# Celsius = (Fahrenheit - 32) * 5/9

print('EX1: Conversor de temperaturas')

while True: # Loop principal
    print('Escolha a sua opção:\nDigite 1 para converter de Fahrenheit para Celsius')
    escolha = input('Digite 2 para converter de Celsius para Fahrenheit: ') #Recebe a string escolha do usuario

    if escolha == '1': # Se a escolha for 1, converte de Fahrenheit para Celsius
        temp_f = float(input('Insira a sua temperatura em Fahrenheit: ')) #Recebe a temperatura em F
        temp_c = (temp_f - 32) * 5/9 #Resultado da conversão
        print(f'A temperatura convertida é de {temp_c} °C') #Exibe a temperatura final convertida em C
        break
    elif escolha == '2': # Se a escolha for 2, converte de Celsius para Fahrenheit
        temp_c = float(input('Insira a sua temperatura em Celsius: ')) #Recebe a temperatura em C
        temp_f = (temp_c * 9/5) + 32 #Resultado da conversão
        print(f'A temperatura convertida é de {temp_f} °F') #Exibe a temperatura final convertida em F
        break
    else:
        print('Escolha inválida, tente novamente:') # Se a escolha for diferente de '1' ou '2', exibe mensagem de erro e repete o loop
    
# 2- Dada a seguinte lista:
# temperaturas = [28.5, 29.1, 27.3, 25.4, 23.9, 22.7, 22.1, 23.5, 24.6, 26.8, 27.9, 28.8]
# Crie um programa que:
# A- Mostre a menor temperatura registrada:
# B- Mostre a maior temperatura registrada:
# C- Mostre a média de temperatura.
# Para este desafio é proibido o uso das funções built-in do python: max(),min() e sum().

temperaturas = [28.5, 29.1, 27.3, 25.4, 23.9, 22.7, 22.1, 23.5, 24.6, 26.8, 27.9, 28.8]
soma = 0 # Inicializa a variável soma com o valor zero
menor = maior = temperaturas[0] #Inicializa as variaveis menor e maior com o primeiro valor da lista

for temperatura in temperaturas: #Itera a lista, para cada temperatura da lista
    soma += temperatura #Para cada temperatura na lista, adiciona o valor a soma de temperaturas

    #Confere se cada temperatura é maior do que o maior valor já registrado, se for, atualiza o maior valor
    if temperatura > maior:
        maior = temperatura
    
    #Confere se cada temperatura é menor do que o menor valor já registrado, se for, atualiza o menor valor
    if temperatura < menor:
        menor = temperatura
    

media = soma/len(temperaturas) #Calcula a media das temperaturas da lista

#Exibe os resultados
print('\nEX2: Comparação de temperaturas')
print(f'A menor temperatura é de: {menor}')
print(f'A maior temperatura é de: {maior}')
print(f'A média das temperaturas da lista é de: {media}')

print('\nEX4: Função conversor de temperaturas')

#Cria a função que recebe como parametros a unidade de medida destino da conversão e o valor a ser convertido
def converter_temp(unidade_final, valor):
    if unidade_final == 'c': # Confere se a unidade final é Celsius
        res = (valor - 32) * 5/9 # Calcula o valor convertido para Celsius

    elif unidade_final == 'f': # Confere se a unidade final é Fahrenheit
        res = (valor * 9/5) + 32 # Calcula o valor convertido para Fahrenheit

    else:
        res = None # Se a unidade final for inválida, retorna None

    return res # Retorna o valor convertido ou None

while True: # Loop principal
    print('Conversor de temperaturas! Escolha a unidade de medida destino da conversão')
    print('Digite \'C\' para converter para Celsius')
    final = input('Digite \'F\' para converter para Fahrenheit: ').lower() #Recebe a unidade de medida destino da conversão.
    temp_inicial = float(input('Insira a sua temperatura a converter: ')) # Recebe a temperatura inicial
    resultado = converter_temp(final, temp_inicial) # Chama a função converter_temp com os parametros unidade final e temperatura inicial
    #Confere se o resultado é válido
    if resultado is None:
        # Sendo inválido, exibe mensagem de erro e repete o loop

    else:
        # Sendo válido, exibe o resultado da conversão e finaliza o programa
